// Función para conectar el dashboard de tu PWA con el backend en la nube async function cargarDatosTrading() { try { // Petición segura a través del proxy CORS hacia tu VM de Oracle Cloud const response = await fetch('https://corsproxy.io/?' + encodeURIComponent('http://193.122.243.181:8080/')); if (!response.ok) { throw new Error('Error en la respuesta del servidor'); } const data = await response.json(); // Actualiza el estado visual en tu PWA if (data.bot_status === "online") { // Aquí puedes actualizar los elementos de tu HTML, por ejemplo: console.log("Estado del Bot: Conectado", data); // Ejemplo para actualizar el balance si tienes un elemento con id 'balance' const elemBalance = document.getElementById('balance'); if (elemBalance) { elemBalance.textContent = `$${data.account.balance.toLocaleString()}`; } // Cambiar indicador de estado a verde si tienes uno const indicador = document.getElementById('bot-status-indicator'); if (indicador) { indicador.className = "text-emerald-400"; // Color verde indicador.textContent = "Online"; } } } catch (error) { console.error("Sin conexión con el backend:", error); // Mantener indicador en rojo o sin conexión si falla const indicador = document.getElementById('bot-status-indicator'); if (indicador) { indicador.className = "text-red-500"; indicador.textContent = "Sin conexión"; } } } // Ejecutar al cargar la página y refrescar cada 5 segundos document.addEventListener('DOMContentLoaded', () => { cargarDatosTrading(); setInterval(cargarDatosTrading, 5000); });
